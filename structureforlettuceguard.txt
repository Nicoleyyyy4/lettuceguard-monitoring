PROJECT OVERVIEW:
LettuceGuard is an IoT Smart Hydroponics Monitoring & Control system for commercial
and home lettuce cultivation. It integrates ESP32 microcontrollers with DHT22,
DS18B20, pH, HC-SR04 ultrasonic, and TDS sensors, storing data securely in Firebase
Authentication, Cloud Firestore, and Firebase Realtime Database. The mobile application
is built using Flutter with Neumorphic Material 3 design.

import 'package:firebase_auth/firebase_auth.dart' as fb_auth;
import 'package:firebase_database/firebase_database.dart';
import 'package:cloud_firestore/cloud_firestore.dart';
import '../models/user_model.dart';

class AuthService {
  final fb_auth.FirebaseAuth _auth = fb_auth.FirebaseAuth.instance;
  final DatabaseReference _dbRef = FirebaseDatabase.instance.ref();
  final FirebaseFirestore _firestore = FirebaseFirestore.instance;

  // Get current authenticated user mapped to UserModel
  UserModel? get currentUser {
    final fbUser = _auth.currentUser;
    if (fbUser == null) return null;
    return UserModel(
      uid: fbUser.uid,
      email: fbUser.email ?? '',
      displayName: fbUser.displayName ?? (fbUser.email?.contains('@') == true ? fbUser.email!.split('@')[0] : 'Operator'),
    );
  }

  // Stream for auth state changes if needed
  Stream<UserModel?> get authStateChanges {
    return _auth.authStateChanges().map((fbUser) {
      if (fbUser == null) return null;
      return UserModel(
        uid: fbUser.uid,
        email: fbUser.email ?? '',
        displayName: fbUser.displayName ?? (fbUser.email?.contains('@') == true ? fbUser.email!.split('@')[0] : 'Operator'),
      );
    });
  }

  // Sign In with Email & Password
  Future<UserModel> signInWithEmail(String email, String password) async {
    try {
      fb_auth.UserCredential credential = await _auth.signInWithEmailAndPassword(
        email: email,
        password: password,
      );

      final user = credential.user!;
      return UserModel(
        uid: user.uid,
        email: user.email ?? email,
        displayName: user.displayName ?? email.split('@')[0],
      );
    } catch (e) {
      rethrow;
    }
  }

  // Register with Name, Email & Password (saves to Auth, Cloud Firestore & Realtime Database)
  Future<UserModel> registerWithEmail(String name, String email, String password) async {
    try {
      // 1. Create user in Firebase Authentication
      fb_auth.UserCredential credential = await _auth.createUserWithEmailAndPassword(
        email: email,
        password: password,
      );

      final user = credential.user!;

      // Update display name in Firebase Auth
      await user.updateDisplayName(name);

      // 2. Save user details to Cloud Firestore in collection 'users' with user's UID as doc ID
      await _firestore.collection('users').doc(user.uid).set({
        'uid': user.uid,
        'name': name,
        'email': email,
        'createdAt': FieldValue.serverTimestamp(),
      });

      // 3. Also sync user profile to Firebase Realtime Database
      await _dbRef.child('users').child(user.uid).set({
        'uid': user.uid,
        'name': name,
        'email': email,
        'createdAt': DateTime.now().toIso8601String(),
      });

      return UserModel(
        uid: user.uid,
        email: user.email ?? email,
        displayName: name,
      );
    } catch (e) {
      rethrow;
    }
  }

  // Sign Out
  Future<void> signOut() async {
    await _auth.signOut();
  }
}

  List<String> _getUnmetPasswordRules(String password) {
    final List<String> unmet = [];
    if (password.length < 8 || password.length > 12) {
      unmet.add('Length must be between 8 and 12 characters.');
    }
    if (!RegExp(r'[A-Z]').hasMatch(password)) {
      unmet.add('Must contain at least 1 Uppercase letter (A-Z).');
    }
    if (!RegExp(r'[0-9]').hasMatch(password)) {
      unmet.add('Must contain at least 1 Number (0-9).');
    }
    if (!RegExp(r'[!@#$%^&*(),.?":{}|<>]').hasMatch(password)) {
      unmet.add('Must contain at least 1 Special character (!@#\$%^&*...).');
    }
    return unmet;
  }

    if (password.isEmpty) {

  Future<void> _handleRegistration() async {

    final email = emailController.text.trim();
    final password = passwordController.text.trim();

    if (name.isEmpty || email.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('Please fill in your name and email address.'),
          backgroundColor: LettuceGuardColors.statusError,
        ),
      );
      return;
    }

    final unmetRules = _getUnmetPasswordRules(password);
    if (unmetRules.isNotEmpty) {
      setState(() {
        _passwordErrorText = unmetRules.first;
      });
      _showPasswordGuideDialog(unmetRules);
      return;
    }

    setState(() {
      _passwordErrorText = null;
      _isLoading = true;
    });

    try {
      // Register in Firebase Auth & write profile to Cloud Firestore (collection 'users')
      await _authService.registerWithEmail(name, email, password);

      if (!mounted) return;

      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text('Account for $name created successfully! Log in to continue.'),
          backgroundColor: LettuceGuardColors.primaryGreen,
        ),
      );

      Navigator.pop(context);
    } catch (e) {
      if (!mounted) return;

      String errorMessage = 'Registration failed. Please check your connection and try again.';
      final errStr = e.toString().toLowerCase();

      if (errStr.contains('email-already-in-use')) {
        errorMessage = 'This email address is already registered.';
      } else if (errStr.contains('invalid-email')) {
        errorMessage = 'Please enter a valid email address.';
      } else if (errStr.contains('weak-password')) {
        errorMessage = 'The password provided is too weak.';
      } else if (errStr.contains('permission-denied')) {
        errorMessage = 'Database permission denied. Check Firebase security rules.';
      }

      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(errorMessage),
          backgroundColor: LettuceGuardColors.statusError,
        ),
      );
    } finally {
      if (mounted) {
        setState(() {
          _isLoading = false;
        });
      }
    }
  }

import '../utils/constants.dart';

class SensorData {
  final double airTemperature;
  final double airHumidity;
  final double waterTemperature;
  final double ph;
  final double tds;
  final double waterLevelPct;
  final double waterLevelCm;

  final bool dht22Error;

  const SensorData({
    required this.airTemperature,

    required this.waterLevelCm,

    this.ultrasonicError = false,
    this.tdsError = false,
    this.adcSaturation = false,
    this.phUncalibrated = false,
  });

  factory SensorData.fromJson(Map<String, dynamic> json) {
    Map<String, dynamic> errors = {};
    if (json['sensorErrors'] is Map) {
      errors = Map<String, dynamic>.from(json['sensorErrors']);
    }

    double levelPct = 0.0;
    double levelCm = 0.0;
    double depthCm = LettuceGuardConstants.defaultReservoirDepthCm;

    if (json['waterLevel'] is Map) {
      final wl = Map<String, dynamic>.from(json['waterLevel']);
      levelPct = (wl['percentage'] as num?)?.toDouble() ?? 0.0;
      levelCm = (wl['distanceCm'] as num?)?.toDouble() ?? 0.0;
      depthCm = (wl['reservoirDepthCm'] as num?)?.toDouble() ?? depthCm;
    } else {
      levelPct = (json['waterLevelPct'] as num?)?.toDouble() ?? 0.0;
      levelCm = (json['waterLevelCm'] as num?)?.toDouble() ?? 0.0;
      depthCm = (json['reservoirDepthCm'] as num?)?.toDouble() ?? depthCm;
    }

    DateTime ts;
    if (json['timestamp'] is int) {
      ts = DateTime.fromMillisecondsSinceEpoch(json['timestamp']);
    } else if (json['timestamp'] is String) {
      ts = DateTime.tryParse(json['timestamp']) ?? DateTime.now();
    } else {
      ts = DateTime.now();
    }

    return SensorData(
      airTemperature: (json['airTemperature'] as num?)?.toDouble() ?? 0.0,
      airHumidity: (json['airHumidity'] as num?)?.toDouble() ?? 0.0,
      waterTemperature: (json['waterTemperature'] as num?)?.toDouble() ?? 0.0,
      ph: (json['ph'] as num?)?.toDouble() ?? 7.0,
      tds: (json['tds'] as num?)?.toDouble() ?? 0.0,
      waterLevelPct: levelPct,
      waterLevelCm: levelCm,
      reservoirDepthCm: depthCm,
      timestamp: ts,
      systemStatus: (json['systemStatus'] as String?) ?? 'OK',
      dht22Error: (errors['dht22Error'] as bool?) ?? false,
      ds18b20Error: (errors['ds18b20Error'] as bool?) ?? false,
      phError: (errors['phError'] as bool?) ?? false,
      ultrasonicError: (errors['ultrasonicError'] as bool?) ?? false,
      tdsError: (errors['tdsError'] as bool?) ?? false,
      adcSaturation: (errors['adcSaturation'] as bool?) ?? false,
      phUncalibrated: (errors['phUncalibrated'] as bool?) ?? false,
    );
  }

  Map<String, dynamic> toJson() {
    return {
      'airTemperature': airTemperature,
      'airHumidity': airHumidity,
      'waterTemperature': waterTemperature,
      'ph': ph,
      'tds': tds,
      'waterLevel': {
        'percentage': waterLevelPct,
        'distanceCm': waterLevelCm,
        'reservoirDepthCm': reservoirDepthCm,
      },
      'systemStatus': calculatedStatus,
      'sensorErrors': {
        'dht22Error': dht22Error,
        'ds18b20Error': ds18b20Error,
        'phError': phError,
        'ultrasonicError': ultrasonicError,
        'tdsError': tdsError,
        'adcSaturation': adcSaturation,
        'phUncalibrated': phUncalibrated,
      },
      'timestamp': timestamp.millisecondsSinceEpoch,
    };
  }

  factory SensorData.mock({

  bool get hasSensorError =>
      dht22Error || ds18b20Error || phError || ultrasonicError || tdsError || adcSaturation || phUncalibrated;

  bool get isAirTempOutOfRange =>
      !dht22Error && (airTemperature < LettuceGuardConstants.minAirTemp || airTemperature > LettuceGuardConstants.maxAirTemp);

  bool get isAirHumidityOutOfRange =>
      !dht22Error && (airHumidity < LettuceGuardConstants.minAirHumidity || airHumidity > LettuceGuardConstants.maxAirHumidity);

  bool get isWaterTempOutOfRange =>
      !ds18b20Error && (waterTemperature < LettuceGuardConstants.minWaterTemp || waterTemperature > LettuceGuardConstants.maxWaterTemp);

  bool get isPhOutOfRange =>
      !phError && !phUncalibrated && (ph < LettuceGuardConstants.minPh || ph > LettuceGuardConstants.maxPh);

  bool get isTdsOutOfRange =>
      !tdsError && (tds < LettuceGuardConstants.minTds || tds > LettuceGuardConstants.maxTds);

  bool get isWaterLevelLow =>
      !ultrasonicError && waterLevelPct < LettuceGuardConstants.minWaterLevelPct;

  bool get hasParameterWarning =>
      isAirTempOutOfRange || isAirHumidityOutOfRange || isWaterTempOutOfRange || isPhOutOfRange || isTdsOutOfRange || isWaterLevelLow;

  String get calculatedStatus {
    if (hasSensorError) return 'ERROR';
    if (hasParameterWarning) return 'WARNING';
    return 'OK';
  }

#include <WiFi.h>
#include <HTTPClient.h>
#include <DHT.h>
#include <OneWire.h>
#include <DallasTemperature.h>
#include <ArduinoJson.h>

const char* WIFI_SSID = "YOUR_WIFI_SSID";
const char* WIFI_PASSWORD = "YOUR_WIFI_PASSWORD";
const char* FIREBASE_HOST = "https://lettuceguard-app-default-rtdb.firebaseio.com";
const char* DEVICE_ID = "lettuceguard_node_01";

#define DHTPIN 4
#define DHTTYPE DHT22
#define ONE_WIRE_BUS 15
#define PH_PIN 34
#define TDS_PIN 35
#define TRIG_PIN 5
#define ECHO_PIN 18

const float RESERVOIR_DEPTH_CM = 30.0;
DHT dht(DHTPIN, DHTTYPE);
OneWire oneWire(ONE_WIRE_BUS);
DallasTemperature ds18b20(&oneWire);

void setup() {
  Serial.begin(115200);
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  dht.begin();
  ds18b20.begin();
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
}

void loop() {
  if (WiFi.status() == WL_CONNECTED) {
    float airTempC = dht.readTemperature();
    float airHumidity = dht.readHumidity();
    ds18b20.requestTemperatures();
    float waterTempC = ds18b20.getTempCByIndex(0);

    StaticJsonDocument<256> doc;
    doc["airTemperature"] = airTempC;
    doc["airHumidity"] = airHumidity;
    doc["waterTemperature"] = waterTempC;

    String jsonString;
    serializeJson(doc, jsonString);

    HTTPClient http;
    http.begin(String(FIREBASE_HOST) + "/devices/" + DEVICE_ID + "/current.json");
    http.addHeader("Content-Type", "application/json");
    http.PUT(jsonString);
    http.end();
  }
  delay(3000);
}
